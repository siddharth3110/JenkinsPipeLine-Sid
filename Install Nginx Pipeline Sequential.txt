

pipeline

{
agent any
	stages
		{
		stage ('install nginx')
			{
			steps 
				{
				sh 'sudo apt update -y'
				sh 'sudo apt install nginx -y'
				}
			}
		
		stage ('update nginx config files')
			{
			steps 
				{
				sh 'echo "This is via pipeline" | sudo tee /var/www/html/index.html'
				//sh 'sudo mv index.html '
				}
			}
		
		stage ('start nginx')
			{
			steps 
				{
				sh 'sudo systemctl enable nginx'
				sh 'sudo systemctl start nginx'
				}
			}
		
		}
}
