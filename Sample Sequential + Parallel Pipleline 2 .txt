//seq+parallel
pipeline
{
    agent any
    stages
    {
        stage ('seq1')
        {
            steps {sh 'echo "This is seqq 1"'}
        }
        
         stage ('seq2')
        {
            steps {sh 'echo "This is seqq 2"'}
        }
        
         stage ('seq3')
        {
            steps {sh 'echo "This is seqq 3"'}
        }
        
        stage ('parallel')
        {
            parallel
            {
                stage ('Build 1.1')
                {
                    steps {sh 'echo "This is build 1.1"'}
                }
                
                 stage ('Build 1.2')
                {
                    steps {sh 'echo "This is build 1.2"'}
                }
                
                 stage ('Build 1.3')
                {
                    steps {sh 'echo "This is build 1.3"'}
                }
            }
        }
    }
}
